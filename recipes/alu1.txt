OUT Y, COUT
BAND 0
qAB = A AND B
qO = A OR B
qX = A XOR B
qU = qX AND CIN
qC = qAB OR qU
qS = qX XOR CIN
qn1a = NOT OP1
qn1b = NOT OP1
qn0 = NOT OP0
m0 = qAB AND qn1a
t0 = m0 AND qn0
m1 = qO AND qn1b
t1 = m1 AND OP0
m2 = qS AND OP1
t2 = m2 AND qn0
m3 = qX AND OP1
t3 = m3 AND OP0
m4 = qC AND OP1
COUT = m4 AND qn0
BAND 1
o1 = t0 OR t1
o2 = o1 OR t2
Y = o2 OR t3

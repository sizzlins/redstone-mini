IN A0, A1, B0, B1, SUB, CI
OUT S0, S1, COUT, DIFF
nSUB = NOT SUB
x0 = B0 XOR SUB
x1 = B1 XOR SUB
c0 = SUB AND CI
p0 = A0 XOR x0
s0 = p0 XOR c0
k0 = A0 AND x0
m0 = p0 AND c0
C1 = k0 OR m0
p1 = A1 XOR x1
s1 = p1 XOR C1
k1 = A1 AND x1
m1 = p1 AND C1
COUT = k1 OR m1
e0 = A0 AND nSUB
f0 = x0 AND SUB
g0 = e0 OR f0
e1 = A1 AND nSUB
f1 = x1 AND SUB
g1 = e1 OR f1
h0 = g0 AND g1
DIFF = h0 OR h0
